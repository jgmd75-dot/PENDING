function kr_r(x, y, w, h) ( gfx_rect(x * sc, y * sc, w * sc, h * sc); );
function kr_txt(x, y, s) ( gfx_x = x * sc; gfx_y = y * sc; gfx_drawstr(s); );
function kr_ctxt(cx, cy, s) local(tw, th) ( gfx_measurestr(s, tw, th); gfx_x = cx * sc - tw * 0.5; gfx_y = cy * sc - th * 0.5; gfx_drawstr(s); );
function kr_rtxt(rx, cy, s) local(tw, th) ( gfx_measurestr(s, tw, th); gfx_x = rx * sc - tw; gfx_y = cy * sc - th * 0.5; gfx_drawstr(s); );
function kr_lbl(x, y, s) ( gfx_setfont(11); ui_col(KRI2); kr_txt(x, y, s); );
function kr_hov(x, y, w, h) ( mouse_x >= x * sc && mouse_x < (x + w) * sc && mouse_y >= y * sc && mouse_y < (y + h) * sc; );
function kr_thick(x1, y1, x2, y2, w) local(dx, dy, l, nx, ny, k) (       // (drawn here: works with any system files)
  dx = x2 - x1; dy = y2 - y1; l = max(sqrt(dx * dx + dy * dy), 0.001); nx = -dy / l; ny = dx / l;
  k = -0.5 * w; loop(floor(w * 2.5) + 1, gfx_line(x1 + nx * k, y1 + ny * k, x2 + nx * k, y2 + ny * k, 1); k += 0.4; );
);
function kr_key(x, y, w, h, lbl, on) local(hov, hit) (
  hov = kr_hov(x, y, w, h); hit = hov && pressed && drag_id == 0; hit ? drag_id = -1;
  ui_col(0x050505); kr_r(x, y, w, h);
  on ? ui_col(KRA) : ui_col(hov ? ui_mix(KRP, 0xFFFFFF, 0.14) : ui_mix(KRP, 0xFFFFFF, 0.07)); kr_r(x + 1, y + 1, w - 2, h - 2);
  on ? ( ui_cola(KRA, 0.25); kr_r(x - 2, y - 2, w + 4, h + 4); ui_cola(0xFFFFFF, 0.3); kr_r(x + 1, y + 1, w - 2, 1); );
  on ? ui_col(KRKT) : ui_col(KRI);
  gfx_setfont(12); kr_ctxt(x + w * 0.5, y + h * 0.5 + 0.5, lbl);
  hit;
);
function kr_dim(x, y, w, h, lbl) ( ui_col(0x050505); kr_r(x, y, w, h); ui_col(ui_mix(KRF, KRP, 0.3)); kr_r(x + 1, y + 1, w - 2, h - 2);
  ui_col(ui_mix(KRI, KRF, 0.75)); gfx_setfont(12); kr_ctxt(x + w * 0.5, y + h * 0.5 + 0.5, lbl); 0; );
function kr_sect(x, y, w, h, name) (
  ui_col(0x040404); kr_r(x, y, w, h); ui_col(KRF); kr_r(x + 1, y + 1, w - 2, h - 2);
  strlen(name) ? kr_lbl(x + 9, y + 6, name);
);
function kr_scale(cx, cy, r, n, nr, labels, minor) local(i, k, k0, len, c, a, ca, sa, m) (
  len = strlen(labels); k0 = 0; i = 0; k = 0;
  minor > 0 ? (
    m = 0; ui_col(KRI2);
    loop((n - 1) * minor + 1,
      a = -0.75 * $pi + m / ((n - 1) * minor) * 1.5 * $pi; ca = sin(a); sa = -cos(a);
      gfx_line((cx + ca * (r + 4)) * sc, (cy + sa * (r + 4)) * sc, (cx + ca * (r + 6)) * sc, (cy + sa * (r + 6)) * sc, 1);
      m += 1;
    );
  );
  gfx_setfont(10);
  loop(len + 1,
    c = k < len ? str_getchar(labels, k) : 124;
    c == 124 ? (
      strcpy_substr(#krv_sl, labels, k0, k - k0);
      a = -0.75 * $pi + i / max(n - 1, 1) * 1.5 * $pi; ca = sin(a); sa = -cos(a);
      ui_col(KRI);
      gfx_line((cx + ca * (r + 3)) * sc, (cy + sa * (r + 3)) * sc, (cx + ca * (r + 7)) * sc, (cy + sa * (r + 7)) * sc, 1);
      strlen(#krv_sl) ? kr_ctxt(cx + ca * (r + nr), cy + sa * (r + nr), #krv_sl);
      i += 1; k0 = k + 1;
    );
    k += 1;
  );
);
function kr_bake(cx, cy, r, a) local(i, t, ca, sa, k) (
  ca = sin(a); sa = -cos(a);
  ui_cola(0x000000, 0.55); gfx_circle(cx + 2.5 * sc, cy + 4.5 * sc, r * 1.02, 1, 1);
  ui_col(0x0A0A0A); gfx_circle(cx, cy, r + 1 * sc, 1, 1);
  ui_col(0x191817); gfx_circle(cx, cy, r, 1, 1);
  i = 0;
  loop(16,
    t = i / 16 * 2 * $pi; k = 0.5 + 0.5 * cos(t + 0.8);
    ui_col(ui_mix(0x0D0C0B, 0x48443F, k * 0.8)); gfx_circle(cx + sin(t) * r * 0.86, cy - cos(t) * r * 0.86, r * 0.15, 1, 1);
    ui_col(0x0D0C0B); gfx_circle(cx + sin(t + 0.196) * r * 0.9, cy - cos(t + 0.196) * r * 0.9, r * 0.075, 1, 1);
    i += 1;
  );
  ui_col(0x090808); gfx_circle(cx, cy, r * 0.70 + 1 * sc, 1, 1);
  ui_col(0x22201E); gfx_circle(cx, cy, r * 0.70, 1, 1);
  ui_cola(0xFFFFFF, 0.08); gfx_circle(cx, cy, r * 0.70, 0, 1);
  ui_col(0x060606); kr_thick(cx + ca * r * 0.18, cy + sa * r * 0.18, cx + ca * r * 0.98, cy + sa * r * 0.98, 4.2 * sc);
  ui_col(KRA); kr_thick(cx + ca * r * 0.22, cy + sa * r * 0.22, cx + ca * r * 0.95, cy + sa * r * 0.95, 2.5 * sc);
);
function kr_scap(cx, cy, r, a, cap) local(ca, sa, k, pc) (
  ca = sin(a); sa = -cos(a);
  ui_cola(0x000000, 0.5); gfx_circle(cx + 1.5 * sc, cy + 2.5 * sc, r, 1, 1);
  ui_col(0x121212); gfx_circle(cx, cy, r, 1, 1);
  ui_cola(0xFFFFFF, 0.12); gfx_circle(cx, cy, r, 0, 1);
  k = r * 0.74;
  ui_col(cap); gfx_circle(cx, cy, k, 1, 1);
  ui_col(ui_mix(cap, 0x000000, 0.35)); gfx_circle(cx, cy, k, 0, 1);
  pc = (cap & 255) + ((cap >> 8) & 255) + ((cap >> 16) & 255) > 380 ? KRPT : 0xF4F0E6;
  ui_col(pc); kr_thick(cx + ca * k * 0.2, cy + sa * k * 0.2, cx + ca * r * 0.94, cy + sa * r * 0.94, 2.6 * sc);
);
krv_small = 0; krv_lg = 0; krv_step = 0; krv_cap = 0;
function kr_kn(v, lo, hi) ( krv_lg ? log(max(v, lo) / lo) / log(hi / lo) : (v - lo) / (hi - lo); );
function kr_kv(n, lo, hi) ( krv_lg ? lo * exp(n * log(hi / lo)) : lo + n * (hi - lo); );
function kr_knob(cx, cy, r, idx, lo, hi, def) local(v, n, dx, dy, hov, did) (
  did = 1700 + idx; v = slider(idx);
  dx = mouse_x - cx * sc; dy = mouse_y - cy * sc; hov = dx * dx + dy * dy <= sqr((r + 5) * sc);
  pressed && hov && drag_id == 0 ? (
    time_precise() - krv_lt < 0.35 && krv_lid == idx ? ( slider(idx) = def; ui_auto(idx); drag_id = -1; krv_lt = 0; ) : (
      drag_id = did; krv_y = mouse_y; krv_lt = time_precise(); krv_lid = idx; krv_n = kr_kn(v, lo, hi);
    );
  );
  drag_id == did ? (
    krv_n = min(max(krv_n + (krv_y - mouse_y) / (((mouse_cap & 8) ? 1200 : (krv_step ? 160 : 240)) * sc), 0), 1); krv_y = mouse_y;
    v = kr_kv(krv_n, lo, hi); krv_step ? v = floor(v + 0.5);
    slider(idx) != v ? ( slider(idx) = v; ui_auto(idx); );
  );
  hov && mouse_wheel != 0 && drag_id == 0 ? (
    krv_step ? slider(idx) = min(max(floor(v + 0.5) + sign(mouse_wheel), lo), hi) : (
      n = min(max(kr_kn(v, lo, hi) + mouse_wheel / 120 * ((mouse_cap & 8) ? 0.005 : 0.025), 0), 1);
      slider(idx) = kr_kv(n, lo, hi);
    );
    ui_auto(idx); mouse_wheel = 0;
  );
  v = slider(idx); n = min(max(kr_kn(v, lo, hi), 0), 1);
  krv_small ? kr_scap(cx * sc, cy * sc, r * sc, -0.75 * $pi + n * 1.5 * $pi, krv_cap ? krv_cap : KRC) : kr_bake(cx * sc, cy * sc, r * sc, -0.75 * $pi + n * 1.5 * $pi);
  hov || drag_id == did ? ( ui_cola(0xFFFFFF, 0.07); gfx_circle(cx * sc, cy * sc, (r + 2) * sc, 1, 1); );
  hov || drag_id == did;
);
function kr_name(cx, cy, act, name, val) ( gfx_setfont(14); ui_col(act ? KRA : KRI); kr_ctxt(cx, cy, act ? val : name); );
// a small knob with a tick ring (or n step ticks) and its name; the value while touched
function kr_small(cx, cy, r, idx, lo, hi, def, nt, name, val) local(i, t, act) (
  ui_cola(KRI, 0.85); i = 0;
  loop(nt, t = -0.75 * $pi + i / (nt - 1) * 1.5 * $pi;
    gfx_line((cx + sin(t) * (r + 4)) * sc, (cy - cos(t) * (r + 4)) * sc, (cx + sin(t) * (r + 8)) * sc, (cy - cos(t) * (r + 8)) * sc, 1); i += 1; );
  krv_small = 1; act = kr_knob(cx, cy, r, idx, lo, hi, def); krv_small = 0;
  strlen(name) ? ( gfx_setfont(11); ui_col(act ? KRA : KRI); kr_ctxt(cx, cy + r + 15, act ? val : name); );
  act;
);
function kr_cell(x, y, w, h, lit, zone) local(c) (
  c = zone == 2 ? KRRED : zone == 1 ? KRAMB : KRCN;
  lit ? ( ui_cola(c, 0.16); kr_r(x - 0.5, y - 1, w + 1, h + 2); ui_col(c); kr_r(x, y, w, h); ui_cola(0xFFFFFF, 0.22); kr_r(x, y, w, 1.5); )
      : ( ui_col(KRGH); kr_r(x, y, w, h); ui_cola(0x000000, 0.25); kr_r(x, y + h - 1, w, 1); );
);
function kr_hbar(x, y, w, h, v) local(n, i, sw, d) (
  n = 32; sw = w / n; i = 0;
  loop(n, d = -48 + (i + 0.5) / n * 48; kr_cell(x + i * sw + 0.5, y, sw - 2, h, v >= d, d > -3 ? 2 : d > -12 ? 1 : 0); i += 1; );
);
function kr_grbar(x, y, w, h, gr) local(n, i, sw, d) (                     // gain reduction: 0 .. -24 dB, from the left
  n = 24; sw = w / n; i = 0;
  loop(n, d = -(i + 0.5); kr_cell(x + i * sw + 0.5, y, sw - 2, h, gr <= d, 1); i += 1; );
);
function kr_window(x, y, w, h) ( ui_col(0x040404); kr_r(x, y, w, h); ui_col(KRW); kr_r(x + 1, y + 1, w - 2, h - 2); );
